Vagrant.configure("2") do |config|
  servers = [
    {
      :hostname => "Server1",
      :box => "stromweld/windows-11",
      :box_version => "202508.09.0",
      :ip => "192.168.56.150" # Changed to match VMware standard subnet
    },
    {
      :hostname => "Server2",
      :box => "stromweld/windows-11",
      :box_version => "202508.09.0",
      :ip => "192.168.56.151" # Changed to match VMware standard subnet
    }
  ]

  servers.each do |machine|
    config.vm.define machine[:hostname] do |node|
      node.vm.box = machine[:box]
      node.vm.box_version = machine[:box_version]
      node.vm.box_architecture = "arm64"
      node.vm.hostname = machine[:hostname]

      # 1. FORCE WINRM COMMUNICATOR
      node.vm.communicator = "winrm"
      node.winrm.username = "vagrant"
      node.winrm.password = "vagrant"
      node.winrm.retry_limit = 30
      node.winrm.retry_delay = 10
      
      # Private network on a unified subnet
      node.vm.network :private_network, ip: machine[:ip]

      # We removed the manual "forwarded_port" line to let Vagrant handle SSH natively

      # 2. FIXED: CHANGED VIRTUALBOX TO VMWARE_DESKTOP
      node.vm.provider :vmware_desktop do |v|
        v.memory = 4096  # Windows 11 strictly requires at least 4GB to initialize properly
        v.cpus = 2       # Gives it enough processing power to actually boot up the WinRM service
        # v.gui = true     # Pops open the window so you can see if Windows is frozen on a setup screen

        # Fixes the network adapter conflicts causing the address timeout
        v.vmx["ethernet0.pcislotnumber"] = "160"
        v.vmx["ethernet1.pcislotnumber"] = "224"
      end
    end
  end
end
